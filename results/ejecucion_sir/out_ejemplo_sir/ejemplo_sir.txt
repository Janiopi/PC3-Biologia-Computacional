// SIR de prueba con LHS
$Initial_conditions
S = 9990;
I = 10;
R = 0;
beta  = 0.3 0.1 0.1 0.5 $U;
gamma = 0.1 0.05 0.05 0.2 $U;
N0    = 10000 9000 9000 11000 $U;
$Equations
N = S+I+R;
S' = -beta*S*I/N;
I' = beta*S*I/N - gamma*I;
R' = gamma*I;
$StartTime 0
$StopTime 100
$timestep 0.05
$timefigs 1
$plot S I R;
$Method $Rk4
$lhs 100 beta gamma N0;
$seed 12345
